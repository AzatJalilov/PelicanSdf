```js
export function createPelicanSdf(canvas) {
  const gl = canvas.getContext("webgl2", {
    antialias: false, alpha: false, depth: false, stencil: false,
    premultipliedAlpha: false, preserveDrawingBuffer: false,
    powerPreference: "high-performance"
  });
  if (!gl) throw new Error("PelicanSDF: WebGL2 is not available");

  const VS = `#version 300 es
void main(){
  vec2 p = vec2(float((gl_VertexID<<1)&2), float(gl_VertexID&2));
  gl_Position = vec4(p*2.0-1.0, 0.0, 1.0);
}`;

  const FS = `...`;

  let prog = null, vao = null, uRes=null, uTime=null, uRo=null, uTa=null, lost=false;

  function compile(type, src){
    const s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if(!gl.getShaderParameter(s, gl.COMPILE_STATUS)){
      const log = gl.getShaderInfoLog(s);
      gl.deleteShader(s);
      throw new Error("PelicanSDF shader compile error: " + log);
    }
    return s;
  }

  function build(){
    const vs = compile(gl.VERTEX_SHADER, VS);
    const fs = compile(gl.FRAGMENT_SHADER, FS);
    const p = gl.createProgram();
    gl.attachShader(p, vs);
    gl.attachShader(p, fs);
    gl.linkProgram(p);
    gl.deleteShader(vs); gl.deleteShader(fs);
    if(!gl.getProgramParameter(p, gl.LINK_STATUS)){
      const log = gl.getProgramInfoLog(p);
      gl.deleteProgram(p);
      throw new Error("PelicanSDF program link error: " + log);
    }
    prog = p;
    uRes = gl.getUniformLocation(p, "u_res");
    uTime = gl.getUniformLocation(p, "u_time");
    uRo = gl.getUniformLocation(p, "u_ro");
    uTa = gl.getUniformLocation(p, "u_ta");
    vao = gl.createVertexArray();
  }
  build();
  ...
}
```
