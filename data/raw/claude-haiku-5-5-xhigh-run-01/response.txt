const VERTEX_SOURCE = `#version 300 es
void main() {
  vec2 p = vec2(float((gl_VertexID << 1) & 2), float(gl_VertexID & 2));
  gl_Position = vec4(p * 2.0 - 1.0, 0.0, 1.0);
}
`;

const FRAGMENT_SOURCE = `#version 300 es
precision highp float;
precision highp int;

uniform vec2 uRes;
uniform float uTime;
uniform vec3 uEye;
uniform vec3 uFwd;
uniform vec3 uRight;
uniform vec3 uUp;
out vec4 fragColor;

const vec3 BB = vec3(-0.02, 0.53, 0.0);
const vec3 RA = vec3(-0.78, 0.595, 0.0);
const vec3 FA = vec3(0.78, 0.595, 0.0);
const vec3 ST = vec3(-0.29, 1.13, 0.0);
const vec3 TT = vec3(-0.23, 1.0, 0.0);
const vec3 HT = vec3(0.52, 1.1, 0.0);
const vec3 HB = vec3(0.6, 0.92, 0.0);

vec3 gPN;
vec3 gPF;
vec3 gAN;
vec3 gAF;
vec3 gKN;
vec3 gKF;

vec2 opU(vec2 a, vec2 b) { return a.x < b.x ? a : b; }

vec2 opS(vec2 a, vec2 b, float k) {
  float h = clamp(0.5 + 0.5 * (b.x - a.x) / k, 0.0, 1.0);
  return vec2(mix(b.x, a.x, h) - k * h * (1.0 - h), h < 0.5 ? b.y : a.y);
}

vec2 rot(vec2 v, float a) {
  float c = cos(a);
  float s = sin(a);
  return vec2(c * v.x - s * v.y, s * v.x + c * v.y);
}

float sdCap(vec3 p, vec3 a, vec3 b, float r) {
  vec3 pa = p - a;
  vec3 ba = b - a;
  float h = clamp(dot(pa, ba) / dot(ba, ba), 0.0, 1.0);
  return length(pa - ba * h) - r;
}

float sdRC(vec3 p, vec3 a, vec3 b, float r1, float r2) {
  vec3 ba = b - a;
  float l2 = dot(ba, ba);
  float rr = r1 - r2;
  float a2 = l2 - rr * rr;
  float il2 = 1.0 / l2;
  vec3 pa = p - a;
  float y = dot(pa, ba);
  float z = y - l2;
  vec3 xv = pa * l2 - ba * y;
  float x2 = dot(xv, xv);
  float y2 = y * y * l2;
  float z2 = z * z * l2;
  float k = sign(rr) * rr * rr * x2;
  if (sign(z) * a2 * z2 > k) return sqrt(x2 + z2) * il2 - r2;
  if (sign(y) * a2 * y2 < k) return sqrt(x2 + y2) * il2 - r1;
  return (sqrt(x2 * a2 * il2) + y * rr) * il2 - r1;
}

float sdFlat(vec3 p, vec3 a, vec3 b, float r1, float r2, float k) {
  return sdRC(vec3(p.x, p.y, p.z * k), a, b, r1, r2) / k;
}

float sdEll(vec3 p, vec3 r) {
  float k0 = length(p / r);
  float k1 = max(length(p / (r * r)), 1e-6);
  return k0 * (k0 - 1.0) / k1;
}

float sdBox(vec3 p, vec3 b) {
  vec3 q = abs(p) - b;
  return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
}

float sdTorusZ(vec3 p, float R, float r) {
  return length(vec2(length(p.xy) - R, p.z)) - r;
}

float sdDiskZ(vec3 p, float R, float h) {
  vec2 d = vec2(length(p.xy) - R, abs(p.z) - h);
  return min(max(d.x, d.y), 0.0) + length(max(d, 0.0));
}

vec2 wheel(vec3 p, vec3 c, float ang) {
  vec3 q = p - c;
  float bd = sdBox(q, vec3(0.62, 0.62, 0.14));
  if (bd > 0.03) return vec2(bd, 0.0);
  vec2 r = vec2(sdTorusZ(q, 0.52, 0.075), 1.0);
  q.xy = rot(q.xy, ang);
  r = opU(r, vec2(sdTorusZ(q, 0.5, 0.022), 2.0));
  r = opU(r, vec2(sdDiskZ(q, 0.065, 0.075), 2.0));
  const float sec = 0.39269908;
  float an = mod(atan(q.y, q.x) + 0.5 * sec, sec) - 0.5 * sec;
  vec2 s = length(q.xy) * vec2(cos(an), sin(an));
  r = opU(r, vec2(sdBox(vec3(s.x - 0.28, s.y, q.z), vec3(0.21, 0.008, 0.012)), 2.0));
  return r;
}

vec2 frameG(vec3 p) {
  float bd = sdBox(p - vec3(0.05, 0.9, 0.0), vec3(0.92, 0.62, 0.3));
  if (bd > 0.03) return vec2(bd, 0.0);
  vec3 pm = vec3(p.x, p.y, abs(p.z));
  vec2 r = vec2(sdCap(p, RA, BB, 0.03), 3.0);
  r = opU(r, vec2(sdCap(p, RA, ST, 0.028), 3.0));
  r = opU(r, vec2(sdCap(p, BB, ST, 0.034), 3.0));
  r = opU(r, vec2(sdCap(p, TT, HT, 0.03), 3.0));
  r = opU(r, vec2(sdCap(p, BB, HB, 0.04), 3.0));
  r = opU(r, vec2(sdCap(p, HB, HT, 0.05), 3.0));
  r = opU(r, vec2(sdCap(pm, vec3(HB.x, HB.y - 0.03, 0.05), vec3(FA.x, FA.y, 0.08), 0.022), 3.0));
  r = opU(r, vec2(sdCap(p, ST, vec3(-0.3, 1.28, 0.0), 0.02), 2.0));
  r = opU(r, vec2(sdEll(p - vec3(-0.31, 1.33, 0.0), vec3(0.19, 0.045, 0.068)), 11.0));
  r = opU(r, vec2(sdCap(p, HT, vec3(0.56, 1.31, 0.0), 0.035), 11.0));
  r = opU(r, vec2(sdCap(p, vec3(0.56, 1.31, -0.22), vec3(0.56, 1.31, 0.22), 0.02), 11.0));
  r = opU(r, vec2(sdCap(pm, vec3(0.56, 1.31, 0.0), vec3(0.5, 1.2, 0.22), 0.022), 11.0));
  r = opU(r, vec2(sdDiskZ(p - vec3(BB.xy, 0.1), 0.1, 0.006), 12.0));
  r = opU(r, vec2(sdCap(p, vec3(BB.xy, -0.12), vec3(BB.xy, 0.12), 0.02), 12.0));
  r = opU(r, vec2(sdCap(p, vec3(BB.xy, 0.1), gPN, 0.016), 12.0));
  r = opU(r, vec2(sdCap(p, vec3(BB.xy, -0.1), gPF, 0.016), 12.0));
  r = opU(r, vec2(sdBox(p - gPN, vec3(0.06, 0.012, 0.035)), 12.0));
  r = opU(r, vec2(sdBox(p - gPF, vec3(0.06, 0.012, 0.035)), 12.0));
  return r;
}

vec2 bird(vec3 p) {
  vec3 q = p - vec3(-0.22, 1.66, 0.0);
  q.xy = rot(q.xy, -0.2);
  vec2 r = vec2(sdEll(q, vec3(0.44, 0.27, 0.27)), 6.0);
  r = opU(r, vec2(sdFlat(p, vec3(-0.5, 1.6, 0.0), vec3(-0.84, 1.46, 0.0), 0.12, 0.05, 2.5), 6.0));
  r = opS(r, vec2(sdRC(p, vec3(0.1, 1.84, 0.0), vec3(0.36, 2.06, 0.0), 0.13, 0.095), 6.0), 0.1);
  r = opS(r, vec2(sdRC(p, vec3(0.36, 2.06, 0.0), vec3(0.5, 2.2, 0.0), 0.095, 0.08), 6.0), 0.05);
  r = opS(r, vec2(sdEll(p - vec3(0.56, 2.27, 0.0), vec3(0.17, 0.155, 0.15)), 6.0), 0.08);
  r = opS(r, vec2(sdEll(p - vec3(0.86, 2.0, 0.0), vec3(0.2, 0.14, 0.11)), 5.0), 0.06);
  r = opU(r, vec2(sdRC(p, vec3(0.66, 2.27, 0.0), vec3(1.38, 2.14, 0.0), 0.095, 0.03), 4.0));
  r = opU(r, vec2(sdRC(p, vec3(0.66, 2.19, 0.0), vec3(1.3, 2.07, 0.0), 0.07, 0.02), 4.0));
  vec3 pm = vec3(p.x, p.y, abs(p.z));
  r = opU(r, vec2(length(pm - vec3(0.6, 2.33, 0.14)) - 0.045, 10.0));
  vec3 w = pm - vec3(-0.08, 1.86, 0.14);
  w.xy = rot(w.xy, 0.22 * sin(uTime * 3.0));
  r = opU(r, vec2(sdFlat(w, vec3(0.0), vec3(-0.72, -0.14, 0.0), 0.15, 0.05, 3.0), 7.0));
  r = opU(r, vec2(sdFlat(w, vec3(-0.44, -0.06, 0.0), vec3(-0.76, -0.14, 0.0), 0.085, 0.05, 3.0), 8.0));
  r = opU(r, vec2(sdRC(pm, vec3(0.0, 1.8, 0.14), vec3(0.5, 1.23, 0.21), 0.07, 0.045), 6.0));
  return r;
}

vec2 riderG(vec3 p) {
  float bd = sdBox(p - vec3(0.3, 1.4, 0.0), vec3(1.2, 1.1, 0.3));
  if (bd > 0.03) return vec2(bd, 0.0);
  vec2 r = bird(p);
  r = opU(r, vec2(sdCap(p, vec3(-0.2, 1.42, 0.14), gKN, 0.075), 9.0));
  r = opU(r, vec2(sdCap(p, gKN, gAN, 0.05), 9.0));
  r = opU(r, vec2(sdCap(p, vec3(-0.2, 1.42, -0.14), gKF, 0.075), 9.0));
  r = opU(r, vec2(sdCap(p, gKF, gAF, 0.05), 9.0));
  r = opU(r, vec2(sdEll(p - gPN - vec3(0.02, 0.035, 0.0), vec3(0.1, 0.035, 0.06)), 9.0));
  r = opU(r, vec2(sdEll(p - gPF - vec3(0.02, 0.035, 0.0), vec3(0.1, 0.035, 0.06)), 9.0));
  return r;
}

vec3 knee(vec3 h, vec3 a) {
  vec2 d2 = a.xy - h.xy;
  float len = length(d2);
  vec2 dir = d2 / max(len, 1e-4);
  float d = clamp(len, 0.1, 0.95);
  float x = (0.0196 + d * d) / (2.0 * d);
  float y = sqrt(max(0.25 - x * x, 0.0));
  vec2 perp = vec2(-dir.y, dir.x);
  if (perp.x < 0.0) perp = -perp;
  return vec3(h.xy + dir * x + perp * y, mix(h.z, a.z, 0.5));
}

vec2 map(vec3 p) {
  vec2 r = vec2(p.y, 13.0);
  r = opU(r, wheel(p, RA, -4.0 * uTime));
  r = opU(r, wheel(p, FA, -4.0 * uTime));
  r = opU(r, frameG(p));
  r = opU(r, riderG(p));
  return r;
}

vec2 march(vec3 ro, vec3 rd) {
  float t = 0.0;
  for (int i = 0; i < 150; i++) {
    vec2 h = map(ro + rd * t);
    if (h.x < 0.0003 + 0.0004 * t) return vec2(t, h.y);
    t += 0.85 * h.x;
    if (t > 40.0) break;
  }
  return vec2(t, -1.0);
}

vec3 calcN(vec3 p, float t) {
  vec2 k = vec2(1.0, -1.0);
  float e = 0.0002 + 0.0001 * t;
  return normalize(
    k.xyy * map(p + k.xyy * e).x +
    k.yyx * map(p + k.yyx * e).x +
    k.yxy * map(p + k.yxy * e).x +
    k.xxx * map(p + k.xxx * e).x);
}

float calcAO(vec3 p, vec3 n) {
  float o = 0.0;
  float s = 1.0;
  for (int i = 0; i < 4; i++) {
    float h = 0.03 + 0.12 * float(i);
    o += (h - map(p + n * h).x) * s;
    s *= 0.9;
  }
  return clamp(1.0 - 2.5 * o, 0.0, 1.0);
}

float shadow(vec3 ro, vec3 rd) {
  float res = 1.0;
  float t = 0.02;
  for (int i = 0; i < 40; i++) {
    float h = map(ro + rd * t).x;
    res = min(res, 8.0 * h / t);
    t += clamp(h, 0.02, 0.25);
    if (res < 0.001 || t > 8.0) break;
  }
  return clamp(res, 0.0, 1.0);
}

vec3 albedo(float m) {
  if (m < 1.5) return vec3(0.05, 0.05, 0.055);
  if (m < 2.5) return vec3(0.8, 0.81, 0.84);
  if (m < 3.5) return vec3(0.82, 0.13, 0.1);
  if (m < 4.5) return vec3(0.98, 0.76, 0.28);
  if (m < 5.5) return vec3(0.97, 0.5, 0.2);
  if (m < 6.5) return vec3(0.96, 0.95, 0.92);
  if (m < 7.5) return vec3(0.78, 0.78, 0.8);
  if (m < 8.5) return vec3(0.14, 0.14, 0.15);
  if (m < 9.5) return vec3(0.95, 0.44, 0.16);
  if (m < 10.5) return vec3(0.03, 0.03, 0.03);
  if (m < 11.5) return vec3(0.12, 0.12, 0.13);
  if (m < 12.5) return vec3(0.28, 0.28, 0.3);
  return vec3(0.5, 0.55, 0.5);
}

float specPow(float m) {
  return (m > 1.5 && m < 2.5) ? 80.0 : (m > 12.5 ? 6.0 : 24.0);
}

float specAmt(float m) {
  return (m > 1.5 && m < 2.5) ? 0.9 : (m > 12.5 ? 0.03 : 0.22);
}

vec3 skyCol(vec3 rd, vec3 L) {
  float y = clamp(rd.y, 0.0, 1.0);
  vec3 c = mix(vec3(0.93, 0.95, 0.97), vec3(0.42, 0.6, 0.86), pow(y, 0.55));
  return c + vec3(1.0, 0.85, 0.6) * (0.35 * pow(max(dot(rd, L), 0.0), 48.0));
}

void main() {
  vec2 uv = (2.0 * gl_FragCoord.xy - uRes) / uRes.y;
  vec3 rd = normalize(uFwd * 1.6 + uRight * uv.x + uUp * uv.y);

  float ph = 2.0 * uTime;
  vec2 cA = BB.xy + 0.17 * vec2(cos(ph), sin(ph));
  vec2 cB = BB.xy - 0.17 * vec2(cos(ph), sin(ph));
  gPN = vec3(cA, 0.2);
  gPF = vec3(cB, -0.2);
  gAN = vec3(cA.x - 0.04, cA.y + 0.12, 0.2);
  gAF = vec3(cB.x - 0.04, cB.y + 0.12, -0.2);
  gKN = knee(vec3(-0.2, 1.42, 0.14), gAN);
  gKF = knee(vec3(-0.2, 1.42, -0.14), gAF);

  vec3 L = normalize(vec3(0.45, 0.85, 0.7));
  vec2 hit = march(uEye, rd);
  vec3 col;

  if (hit.y < 0.0) {
    col = skyCol(rd, L);
  } else {
    vec3 p = uEye + rd * hit.x;
    vec3 n = calcN(p, hit.x);
    float m = hit.y;
    vec3 al = albedo(m);
    if (m > 12.5) {
      float dash = step(0.5, fract((p.x + 2.38 * uTime) / 1.2)) * step(abs(p.z + 1.1), 0.035);
      al = mix(vec3(0.5, 0.54, 0.5), vec3(0.92, 0.92, 0.86), dash);
    }
    float occ = calcAO(p, n);
    float sh = shadow(p + n * 0.002, L);
    float dif = max(dot(n, L), 0.0) * sh;
    float skyL = 0.5 + 0.5 * n.y;
    float bounce = clamp(dot(n, normalize(vec3(-L.x, -0.4, -L.z))), 0.0, 1.0);
    vec3 lin = vec3(1.0, 0.96, 0.9) * dif * 1.9
             + vec3(0.55, 0.65, 0.8) * skyL * occ * 0.75
             + vec3(0.45, 0.38, 0.32) * bounce * occ * 0.3;
    vec3 hv = normalize(L - rd);
    float sp = pow(max(dot(n, hv), 0.0), specPow(m)) * specAmt(m) * dif;
    col = al * lin + vec3(sp);
    float fr = pow(1.0 - max(dot(n, -rd), 0.0), 4.0);
    col += fr * 0.12 * occ * vec3(0.85, 0.92, 1.0);
    col = mix(col, skyCol(rd, L), 1.0 - exp(-hit.x * 0.03));
  }

  col = pow(clamp(col, 0.0, 1.0), vec3(0.4545));
  fragColor = vec4(col, 1.0);
}
`;

const UNIFORM_NAMES = ['uRes', 'uTime', 'uEye', 'uFwd', 'uRight', 'uUp'];

export function createPelicanSdf(canvas) {
  if (!canvas || typeof canvas.getContext !== 'function') {
    throw new Error('createPelicanSdf: a canvas element is required.');
  }

  const PITCH_MAX = 1.45;
  const DIST_MIN = 0.9;
  const DIST_MAX = 9.0;
  const FAR_FOCUS = [0.15, 1.0, 0.0];
  const NEAR_FOCUS = [0.35, 1.3, 0.0];

  const clamp = (v, lo, hi) => (v < lo ? lo : v > hi ? hi : v);
  const cross = (a, b) => [
    a[1] * b[2] - a[2] * b[1],
    a[2] * b[0] - a[0] * b[2],
    a[0] * b[1] - a[1] * b[0],
  ];
  const unit = (a) => {
    const l = Math.hypot(a[0], a[1], a[2]) || 1;
    return [a[0] / l, a[1] / l, a[2] / l];
  };
  const focusFor = (d) => {
    const x = clamp((d - 3.6) / (1.5 - 3.6), 0, 1);
    const s = x * x * (3 - 2 * x);
    return FAR_FOCUS.map((f, i) => f + (NEAR_FOCUS[i] - f) * s);
  };

  let yaw = 0.6;
  let pitch = 0.25;
  let dist = 4.2;

  let gl = null;
  let program = null;
  let vao = null;
  let uniforms = {};
  let contextLost = false;
  let disposed = false;
  const pointers = new Map();
  let pinchPrev = 0;

  const compileShader = (type, source, label) => {
    const shader = gl.createShader(type);
    gl.shaderSource(shader, source);
    gl.compileShader(shader);
    if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
      const log = gl.getShaderInfoLog(shader) || 'no log';
      gl.deleteShader(shader);
      throw new Error(`createPelicanSdf: ${label} shader failed to compile: ${log}`);
    }
    return shader;
  };

  const buildProgram = () => {
    const vs = compileShader(gl.VERTEX_SHADER, VERTEX_SOURCE, 'vertex');
    const fs = compileShader(gl.FRAGMENT_SHADER, FRAGMENT_SOURCE, 'fragment');
    const prog = gl.createProgram();
    gl.attachShader(prog, vs);
    gl.attachShader(prog, fs);
    gl.linkProgram(prog);
    gl.deleteShader(vs);
    gl.deleteShader(fs);
    if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) {
      const log = gl.getProgramInfoLog(prog) || 'no log';
      gl.deleteProgram(prog);
      throw new Error(`createPelicanSdf: program failed to link: ${log}`);
    }
    program = prog;
    vao = gl.createVertexArray();
    uniforms = {};
    for (const name of UNIFORM_NAMES) uniforms[name] = gl.getUniformLocation(program, name);
  };

  const camera = () => {
    const cp = Math.cos(pitch);
    const dir = [Math.sin(yaw) * cp, Math.sin(pitch), Math.cos(yaw) * cp];
    const focus = focusFor(dist);
    const eye = [
      focus[0] + dir[0] * dist,
      focus[1] + dir[1] * dist,
      focus[2] + dir[2] * dist,
    ];
    const fwd = [-dir[0], -dir[1], -dir[2]];
    const right = unit(cross(fwd, [0, 1, 0]));
    const up = cross(right, fwd);
    return { eye, fwd, right, up };
  };

  const pinchSpan = () => {
    const v = [...pointers.values()];
    return v.length < 2 ? 0 : Math.hypot(v[0].x - v[1].x, v[0].y - v[1].y);
  };

  const onContextLost = (event) => {
    event.preventDefault();
    contextLost = true;
  };

  const onContextRestored = () => {
    try {
      buildProgram();
      contextLost = false;
    } catch (err) {
      contextLost = true;
    }
  };

  const onPointerDown = (e) => {
    e.preventDefault();
    if (typeof canvas.focus === 'function') canvas.focus({ preventScroll: true });
    try {
      canvas.setPointerCapture(e.pointerId);
    } catch (_) {
      /* ignore */
    }
    pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
    if (pointers.size === 2) pinchPrev = pinchSpan();
  };

  const onPointerMove = (e) => {
    const p = pointers.get(e.pointerId);
    if (!p) return;
    e.preventDefault();
    const dx = e.clientX - p.x;
    const dy = e.clientY - p.y;
    p.x = e.clientX;
    p.y = e.clientY;
    if (pointers.size === 1) {
      yaw += dx * 0.008;
      pitch = clamp(pitch + dy * 0.006, -PITCH_MAX, PITCH_MAX);
    } else {
      const span = pinchSpan();
      if (pinchPrev > 0 && span > 0) {
        dist = clamp(dist * (pinchPrev / span), DIST_MIN, DIST_MAX);
      }
      pinchPrev = span;
    }
  };

  const onPointerUp = (e) => {
    if (!pointers.delete(e.pointerId)) return;
    try {
      canvas.releasePointerCapture(e.pointerId);
    } catch (_) {
      /* ignore */
    }
    pinchPrev = pointers.size === 2 ? pinchSpan() : 0;
  };

  const onWheel = (e) => {
    e.preventDefault();
    let dy = e.deltaY;
    if (e.deltaMode === 1) dy *= 16;
    else if (e.deltaMode === 2) dy *= 400;
    dist = clamp(dist * Math.exp(dy * 0.0012), DIST_MIN, DIST_MAX);
  };

  const onKeyDown = (e) => {
    let used = true;
    switch (e.key) {
      case 'ArrowLeft':
        yaw -= 0.1;
        break;
      case 'ArrowRight':
        yaw += 0.1;
        break;
      case 'ArrowUp':
        pitch = clamp(pitch + 0.08, -PITCH_MAX, PITCH_MAX);
        break;
      case 'ArrowDown':
        pitch = clamp(pitch - 0.08, -PITCH_MAX, PITCH_MAX);
        break;
      case '+':
      case '=':
      case 'Add':
        dist = clamp(dist * 0.9, DIST_MIN, DIST_MAX);
        break;
      case '-':
      case '_':
      case 'Subtract':
        dist = clamp(dist / 0.9, DIST_MIN, DIST_MAX);
        break;
      default:
        used = false;
    }
    if (used) e.preventDefault();
  };

  const preventDefault = (e) => e.preventDefault();

  gl = canvas.getContext('webgl2', {
    alpha: false,
    antialias: false,
    depth: false,
    stencil: false,
    preserveDrawingBuffer: true,
    premultipliedAlpha: false,
    powerPreference: 'high-performance',
  });
  if (!gl) {
    throw new Error('createPelicanSdf: WebGL2 is not available for this canvas.');
  }
  buildProgram();

  canvas.tabIndex = 0;
  canvas.style.touchAction = 'none';

  const listeners = [
    ['webglcontextlost', onContextLost, false],
    ['webglcontextrestored', onContextRestored, false],
    ['pointerdown', onPointerDown, false],
    ['pointermove', onPointerMove, false],
    ['pointerup', onPointerUp, false],
    ['pointercancel', onPointerUp, false],
    ['lostpointercapture', onPointerUp, false],
    ['wheel', onWheel, { passive: false }],
    ['keydown', onKeyDown, false],
    ['touchstart', preventDefault, { passive: false }],
    ['touchmove', preventDefault, { passive: false }],
  ];
  for (const [type, fn, opts] of listeners) canvas.addEventListener(type, fn, opts);

  const render = (timeSeconds) => {
    if (disposed || contextLost || !gl || !program || gl.isContextLost()) return;
    const time = Number.isFinite(timeSeconds) ? timeSeconds : 0;
    const w = gl.drawingBufferWidth;
    const h = gl.drawingBufferHeight;
    const cam = camera();
    gl.viewport(0, 0, w, h);
    gl.useProgram(program);
    gl.bindVertexArray(vao);
    gl.uniform2f(uniforms.uRes, w, h);
    gl.uniform1f(uniforms.uTime, time);
    gl.uniform3f(uniforms.uEye, cam.eye[0], cam.eye[1], cam.eye[2]);
    gl.uniform3f(uniforms.uFwd, cam.fwd[0], cam.fwd[1], cam.fwd[2]);
    gl.uniform3f(uniforms.uRight, cam.right[0], cam.right[1], cam.right[2]);
    gl.uniform3f(uniforms.uUp, cam.up[0], cam.up[1], cam.up[2]);
    gl.drawArrays(gl.TRIANGLES, 0, 3);
  };

  const setView = (nextYaw, nextPitch, nextDistance) => {
    if (Number.isFinite(nextYaw)) yaw = nextYaw;
    if (Number.isFinite(nextPitch)) pitch = clamp(nextPitch, -PITCH_MAX, PITCH_MAX);
    if (Number.isFinite(nextDistance)) dist = clamp(nextDistance, DIST_MIN, DIST_MAX);
  };

  const getView = () => ({ yaw, pitch, distance: dist });

  const dispose = () => {
    if (disposed) return;
    disposed = true;
    for (const [type, fn, opts] of listeners) canvas.removeEventListener(type, fn, opts);
    pointers.clear();
    if (gl && !gl.isContextLost()) {
      if (program) gl.deleteProgram(program);
      if (vao) gl.deleteVertexArray(vao);
    }
    program = null;
    vao = null;
  };

  return { render, setView, getView, dispose };
}
